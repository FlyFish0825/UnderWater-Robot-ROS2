cmake_minimum_required(VERSION 3.16)
project(sensors_cpp)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp ament_index_cpp std_srvs geometry_msgs underwater_interfaces msg_pkg)
  find_package(${dep} REQUIRED)
endforeach()
add_library(compat_runtime SHARED src/entry.cpp src/backend.cpp src/telemetry.cpp src/manual.cpp)
get_filename_component(ROBOT_WORKSPACE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
target_compile_definitions(compat_runtime PRIVATE DEFAULT_RECORD_DIR="${ROBOT_WORKSPACE_ROOT}/Recodes")
target_compile_features(compat_runtime PUBLIC cxx_std_17)
target_compile_options(compat_runtime PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(compat_runtime PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(compat_runtime rclcpp ament_index_cpp std_srvs geometry_msgs underwater_interfaces msg_pkg)
add_executable(backend_host src/backend_host.cpp)
target_link_libraries(backend_host compat_runtime)
install(TARGETS backend_host DESTINATION lib/${PROJECT_NAME})
foreach(entry depth sensors_sub sensors_sub_jilushuju camera_display camera_save camera_ros camera_parallel_ros auto_control auto_control_new semi_control semi_control_fc semi_control_fcadrc)
  add_executable(${entry} src/main.cpp)
  target_compile_definitions(${entry} PRIVATE COMPAT_ENTRY="${entry}")
  target_link_libraries(${entry} compat_runtime)
  install(TARGETS ${entry} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(TARGETS compat_runtime EXPORT compatTargets LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(compatTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp ament_index_cpp std_srvs geometry_msgs underwater_interfaces msg_pkg)
if(BUILD_TESTING)
  add_executable(compat_probe ../../tests/integration/compat_probe.cpp)
  target_link_libraries(compat_probe compat_runtime util)
  install(TARGETS compat_probe DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
