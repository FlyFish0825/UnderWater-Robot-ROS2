cmake_minimum_required(VERSION 3.16)
project(imu_py)
find_package(ament_cmake REQUIRED)
find_package(sensors_cpp REQUIRED)
foreach(entry imu_publisher)
  add_executable(${entry} src/main.cpp)
  target_compile_definitions(${entry} PRIVATE COMPAT_ENTRY="${entry}")
  target_link_libraries(${entry} sensors_cpp::compat_runtime)
  install(TARGETS ${entry} DESTINATION lib/${PROJECT_NAME})
endforeach()
ament_package()
